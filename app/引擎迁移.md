## 项目概述

本项目旨在将原始的大型游戏引擎文件（`engine.ts`，9700+行）拆分为模块化、可维护的组件架构。通过领域驱动设计（DDD）原则，将游戏逻辑按功能领域进行分离，提高代码的可读性、可测试性和可扩展性。

**项目状态**：所有核心游戏系统迁移完成，地面阻挡线功能完整实现，单元测试和集成测试全部通过，进入性能优化和发布准备阶段

### 4. 迁移策略总结

#### 优先级顺序
1. **高级优先级**：核心游戏循环、输入处理、碰撞检测
2. **中级优先级**：经济系统、波次管理、武器系统
3. **低级优先级**：UI渲染、特效、音效系统

#### 迁移方法
1. **逐步替换法**：创建新模块 → 实现核心逻辑 → 集成测试 → 清理旧代码
2. **并行运行法**：新旧系统并行，确保平滑过渡
3. **功能优先法**：按重要性排序迁移顺序

#### 质量保证
- 每次迁移后都要进行测试
- 保证系统可靠性和安全性
- 确保后续新道具的可扩展性
- 最终实现完全脱离原始引擎

```typescript
// RoachManager.ts中的关键实现
getGroundBoundsAtY(y: number): [number, number] {
  const [farL, farLY, farR, farRY, midL, midLY, midR, midRY, nearL, nearR, nearY] = SCENE_GROUND_BOUNDS[this.config.currentScene];
  const clampedY = Math.min(nearY, Math.max(Math.min(farLY, farRY), y));

  // 左侧：2段（远→中→近）
  let leftX: number;
  if (clampedY >= midLY) {
    // 中段和近段之间（下半部分）
    const t = (clampedY - midLY) / (nearY - midLY);
    leftX = midL + (nearL - midL) * t;
  } else {
    // 远段和中段之间（上半部分）
    const t = (clampedY - farLY) / (midLY - farLY);
    leftX = farL + (midL - farL) * t;
  }

  // 右侧：2段（远→中→近）
  let rightX: number;
  if (clampedY >= midRY) {
    // 中段和近段之间（下半部分）
    const t = (clampedY - midRY) / (nearY - midRY);
    rightX = midR + (nearR - midR) * t;
  } else {
    // 远段和中段之间（上半部分）
    const t = (clampedY - farRY) / (midRY - farRY);
    rightX = farR + (midR - farR) * t;
  }

  return [leftX, rightX];
}
```

```typescript
/**
 * 动态粒子限制器
 * @description 根据当前性能自动调整粒子数量限制
 */
export class DynamicParticleLimiter {
  /** 当前粒子数量限制 */
  private currentLimit: number;
  /** 基础限制（性能良好时） */
  private baseLimit: number;
  /** 最小限制（性能较差时） */
  private minLimit: number;
  /** 性能阈值（FPS低于此值时开始限制） */
  private performanceThreshold: number;
  
  /**
   * 创建动态粒子限制器
   * @param config 配置参数
   */
  constructor(config: DynamicParticleLimiterConfig) {
    this.baseLimit = config.baseLimit || 500;
    this.minLimit = config.minLimit || 100;
    this.performanceThreshold = config.performanceThreshold || 30;
    this.currentLimit = this.baseLimit;
  }
  
  /**
   * 根据当前性能更新粒子限制
   * @param currentFps 当前帧率
   * @param currentParticleCount 当前粒子数量
   * @returns 新的粒子数量限制
   */
  updateLimit(currentFps: number, currentParticleCount: number): number {
    if (currentFps < this.performanceThreshold) {
      // 性能较差，减少粒子限制
      const reductionFactor = (this.performanceThreshold - currentFps) / this.performanceThreshold;
      this.currentLimit = Math.max(
        this.minLimit,
        Math.floor(this.baseLimit * (1 - reductionFactor * 0.5))
      );
    } else {
      // 性能良好，逐渐恢复粒子限制
      this.currentLimit = Math.min(
        this.baseLimit,
        this.currentLimit + Math.floor((this.baseLimit - this.currentLimit) * 0.1)
      );
    }
    
    return this.currentLimit;
  }
}
```

- **DefenseCheckSystemConfig接口**：定义系统配置和事件回调

#### 2. 核心算法实现
```typescript
/**
 * 检查防御线突破
 * @description 检测蟑螂是否突破防线并计算伤害
 * @param roaches - 蟑螂数组
 * @returns 是否触发游戏失败
 */
checkDefense(roaches: Roach[]): boolean {
  if (this.config.gameState !== GameState.PLAYING) {
    return false;
  }

  let gameDefeatTriggered = false;
  
  // 从后向前遍历，确保正确移除元素
  for (let i = roaches.length - 1; i >= 0; i--) {
    const roach = roaches[i];
    if (!roach || roach.state !== RoachState.ALIVE) continue;

    const roachSize = this.getRoachSize(roach.type);
    const roachBottom = roach.y + roachSize * 0.4;
    
    if (roachBottom >= this.config.defenseLineY) {
      // 处理特殊蟑螂类型
      if (roach.type === RoachType.SUICIDE) {
        this.handleSuicideExplosion(roach, i, roaches);
        continue;
      }

      // 计算伤害
      const damage = this.calculateBreachDamage(roach);
      
      if (damage > 0) {
        // 更新防御生命值
        this.config.defenseHp = Math.max(0, this.config.defenseHp - damage);
        
        // 触发事件回调
        this.config.onDefenseUpdate?.(this.config.defenseHp, this.config.maxDefenseHp);
        this.config.onAddFloatingText?.(`-${damage}`, roach.x, roach.y, '#ff0000');
        this.config.onPlayAudio?.('defense_hit');
        this.config.onTriggerVibration?.(200);
        this.config.onScreenShake?.(0.3, 200);
        
        // 检查游戏失败条件
        if (this.config.defenseHp <= 0) {
          gameDefeatTriggered = true;
          this.triggerGameDefeat();
        }
      }
      
      // 移除突破防线的蟑螂
      roaches.splice(i, 1);
    }
  }
  
  return gameDefeatTriggered;
}
}
```
