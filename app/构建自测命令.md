## 构建命令
# TypeScript 类型检查（不生成输出，仅验证类型）
npx tsc --noEmit

# Vite 生产构建
npx vite build

# 预览生产构建
npx vite preview

## 开发服务器
# 启动开发服务器（默认端口 5173，仅本地）
npx vite

# 启动开发服务器并暴露到局域网（手机可访问）
npx vite --host 0.0.0.0

# 指定端口启动
npx vite --host 0.0.0.0 --port 5173

## 测试端口命令
# 查看端口占用情况（Windows）
netstat -ano | findstr :5173

# 查看占用端口的进程
Get-Process -Id (Get-NetTCPConnection -LocalPort 5173).OwningProcess -ErrorAction SilentlyContinue

# 终止占用端口的进程（如需要）
taskkill /PID <进程ID> /F


## 一键组合命令

# 先类型检查，再构建
npx tsc --noEmit && npx vite build

# 先构建，再启动局域网预览
npx vite build && npx vite preview --host 0.0.0.0
